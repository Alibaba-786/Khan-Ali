import requests, re, time, random, string, os
from faker import Faker
from bs4 import BeautifulSoup
from fake_useragent import UserAgent

fake = Faker()
ua = UserAgent()

W = "\x1b[97m"
G = "\x1b[38;5;46m"
R = "\x1b[38;5;196m"
X = f"{W}<{R}•{W}>"

def gen_email():
    name = ''.join(random.choices(string.ascii_lowercase + string.digits, k=10))
    domain = random.choice(['1secmail.com', '1secmail.net'])
    return f"{name}@{domain}", name, domain

def get_otp(login, domain):
    for _ in range(15):
        time.sleep(2)
        try:
            inbox = requests.get(f"https://www.1secmail.com/api/v1/?action=getMessages&login={login}&domain={domain}").json()
            for mail in inbox:
                mail_id = mail['id']
                content = requests.get(f"https://www.1secmail.com/api/v1/?action=readMessage&login={login}&domain={domain}&id={mail_id}").json()
                match = re.search(r'FB-(\d+)', content['body'])
                if match:
                    return match.group(1)
        except:
            pass
    return None

def gen_useragent():
    return ua.random

def linex(): print(f"{W}" + "—" * 55)

def banner():
    os.system("clear")
    print(f"{X} FACEBOOK AUTO ID CREATOR v3")
    print(f"{X} Coded by {G}Khan Alibaba 💻")
    linex()

def extract_form(html):
    soup = BeautifulSoup(html, 'html.parser')
    data = {}
    for inp in soup.find_all("input"):
        name = inp.get("name")
        value = inp.get("value", "")
        if name:
            data[name] = value
    return data

def save_success(uid, pwd, cookie):
    with open("/sdcard/SUCCESS-OK-ID.txt", "a") as f:
        f.write(f"{uid}|{pwd}|{cookie}\n")

def create_account(pwd):
    ses = requests.Session()
    ses.headers.update({"User-Agent": gen_useragent()})

    try:
        html = ses.get("https://x.facebook.com/reg").text
    except:
        print(f"{X} {R}Failed to reach Facebook.")
        return

    form = extract_form(html)

    first = fake.first_name()
    last = fake.last_name()
    email, login, domain = gen_email()
    bday = random.randint(1, 28)
    bmonth = random.randint(1, 12)
    byear = random.randint(1992, 2004)

    print(f"{X} NAME  - {G}{first} {last}")
    print(f"{X} EMAIL - {G}{email}")

    payload = {
        'ccp': "2",
        'reg_instance': form.get("reg_instance", ""),
        'submission_request': "true",
        'reg_impression_id': form.get("reg_impression_id", ""),
        'ns': "1",
        'app_id': "103",
        'logger_id': form.get("logger_id", ""),
        'firstname': first,
        'lastname': last,
        'birthday_day': bday,
        'birthday_month': bmonth,
        'birthday_year': byear,
        'reg_email__': email,
        'sex': "2",
        'reg_passwd__': pwd,
        'encpass': f"#PWD_BROWSER:0:{int(time.time())}:{pwd}",
        'submit': "Sign Up",
        'fb_dtsg': form.get("fb_dtsg", ""),
        'jazoest': form.get("jazoest", ""),
        'lsd': form.get("lsd", ""),
    }

    headers = {
        'User-Agent': gen_useragent(),
        'Referer': 'https://x.facebook.com/reg',
        'Origin': 'https://x.facebook.com',
        'Content-Type': 'application/x-www-form-urlencoded',
        'Accept-Language': 'en-US,en;q=0.9',
        'Accept': '*/*',
    }

    try:
        reg = ses.post("https://x.facebook.com/reg/submit/", data=payload, headers=headers)
    except:
        print(f"{X} {R}Form submit failed.")
        return

    cookies = ses.cookies.get_dict()

    if "c_user" in cookies:
        uid = cookies['c_user']
        print(f"{X} FB UID - {G}{uid}")
        print(f"{X} WAITING FOR OTP CODE...")

        otp = get_otp(login, domain)
        if otp:
            print(f"{X} OTP RECEIVED - {G}{otp}")
            confirm_url = "https://m.facebook.com/confirmation_cliff/"
            confirm_payload = {
                'fb_dtsg': form.get("fb_dtsg", ""),
                'jazoest': form.get("jazoest", ""),
                'lsd': form.get("lsd", ""),
                '__user': uid
            }
            confirm_params = {
                'contact': email,
                'type': "submit",
                'is_soft_cliff': "false",
                'medium': "email",
                'code': otp
            }
            ses.post(confirm_url, data=confirm_payload, params=confirm_params)
            cookie_str = "; ".join([f"{k}={v}" for k, v in cookies.items()])
            save_success(uid, pwd, cookie_str)
            print(f"{X} ✅ SUCCESS - {G}{uid}|{pwd}")
        else:
            print(f"{X} ❌ OTP FAILED - ID DISABLED")
    else:
        print(f"{X} ❌ ACCOUNT FAILED / CHECKPOINTED")
    linex()

def main():
    banner()
    try:
        total = int(input(f"{X} How many accounts to create? "))
    except:
        total = 1

    pwd = input(f"{X} Enter password for all accounts (leave blank = random): ")
    if not pwd:
        pwd = ''.join(random.choices(string.ascii_letters + string.digits, k=12)) + "@123"

    linex()
    for i in range(total):
        print(f"{X} Creating account {i+1}")
        create_account(pwd)
        delay = random.randint(1, 5)
        print(f"{X} Sleeping for {delay}s before next...")
        time.sleep(delay)

if __name__ == "__main__":
    main()
